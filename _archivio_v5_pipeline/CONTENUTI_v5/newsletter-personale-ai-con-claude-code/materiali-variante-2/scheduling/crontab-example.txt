# Schedulazione newsletter.py — Variante 2
#
# Sostituisci /path/to/materiali-variante-2 col path reale della tua cartella.

# ====== MAC / LINUX (cron) ======
# crontab -e, poi incolla UNA di queste righe:

# Ogni giorno alle 8:00
0 8 * * * cd /path/to/materiali-variante-2 && /path/to/materiali-variante-2/.venv/bin/python newsletter.py >> /tmp/newsletter.log 2>&1

# Solo nei giorni lavorativi
0 8 * * 1-5 cd /path/to/materiali-variante-2 && /path/to/materiali-variante-2/.venv/bin/python newsletter.py >> /tmp/newsletter.log 2>&1


# ====== WINDOWS (Utilità di pianificazione) ======
# Crea attività → Trigger giornaliero ore 8:00 → Azione "Avvia programma":
#   Programma: C:\path\to\materiali-variante-2\.venv\Scripts\python.exe
#   Argomenti: newsletter.py
#   Inizio in: C:\path\to\materiali-variante-2


# ====== RASPBERRY PI / SERVER (systemd timer) ======
# /etc/systemd/system/newsletter.service:
#   [Service]
#   Type=oneshot
#   WorkingDirectory=/path/to/materiali-variante-2
#   ExecStart=/path/to/materiali-variante-2/.venv/bin/python newsletter.py
#
# /etc/systemd/system/newsletter.timer:
#   [Timer]
#   OnCalendar=*-*-* 08:00:00
#   Persistent=true
#   [Install]
#   WantedBy=timers.target
#
# Attiva: sudo systemctl enable --now newsletter.timer
