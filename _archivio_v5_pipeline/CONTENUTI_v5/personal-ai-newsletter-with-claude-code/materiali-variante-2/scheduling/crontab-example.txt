# Scheduling newsletter.py — Variant 2
#
# Replace /path/to/materiali-variante-2 with the real path of your folder.

# ====== MAC / LINUX (cron) ======
# crontab -e, then paste ONE of these lines:

# Every day at 8:00
0 8 * * * cd /path/to/materiali-variante-2 && /path/to/materiali-variante-2/.venv/bin/python newsletter.py >> /tmp/newsletter.log 2>&1

# Weekdays only
0 8 * * 1-5 cd /path/to/materiali-variante-2 && /path/to/materiali-variante-2/.venv/bin/python newsletter.py >> /tmp/newsletter.log 2>&1


# ====== WINDOWS (Task Scheduler) ======
# Create task → Daily trigger at 8:00 AM → Action "Start a program":
#   Program: C:\path\to\materiali-variante-2\.venv\Scripts\python.exe
#   Arguments: newsletter.py
#   Start in: C:\path\to\materiali-variante-2


# ====== RASPBERRY PI / SERVER (systemd timer) ======
# /etc/systemd/system/newsletter.service:
#   [Service]
#   Type=oneshot
#   WorkingDirectory=/path/to/materiali-variante-2
#   ExecStart=/path/to/materiali-variante-2/.venv/bin/python newsletter.py
#
# /etc/systemd/system/newsletter.timer:
#   [Timer]
#   OnCalendar=*-*-* 08:00:00
#   Persistent=true
#   [Install]
#   WantedBy=timers.target
#
# Enable: sudo systemctl enable --now newsletter.timer


# ====== NOTES ======
# - Check venv python: which python (Mac/Linux), where python (Windows)
# - Log in /tmp/newsletter.log for debugging
# - On Mac, cron requires Full Disk Access in Privacy & Security
