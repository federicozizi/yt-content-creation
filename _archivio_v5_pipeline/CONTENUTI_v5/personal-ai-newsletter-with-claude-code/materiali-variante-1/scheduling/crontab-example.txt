# Scheduling newsletter.py — examples for Mac/Linux/Windows
#
# Replace /path/to/materiali-variante-1 with the real path of your folder.
# Replace the venv Python path too.

# ====== MAC / LINUX (cron) ======
# Open crontab with: crontab -e
# Then paste ONE of these lines:

# Every day at 8:00
0 8 * * * cd /path/to/materiali-variante-1 && /path/to/materiali-variante-1/.venv/bin/python newsletter.py >> /tmp/newsletter.log 2>&1

# Weekdays only
0 8 * * 1-5 cd /path/to/materiali-variante-1 && /path/to/materiali-variante-1/.venv/bin/python newsletter.py >> /tmp/newsletter.log 2>&1

# Every 3 hours (for very active sources)
0 */3 * * * cd /path/to/materiali-variante-1 && /path/to/materiali-variante-1/.venv/bin/python newsletter.py >> /tmp/newsletter.log 2>&1


# ====== WINDOWS (Task Scheduler) ======
# Open "Task Scheduler" → Create task → Daily trigger at 8:00 AM
# Action "Start a program":
#   Program: C:\path\to\materiali-variante-1\.venv\Scripts\python.exe
#   Arguments: newsletter.py
#   Start in: C:\path\to\materiali-variante-1


# ====== SERVER (systemd timer, modern alternative to cron) ======
# File: /etc/systemd/system/newsletter.service
#   [Unit]
#   Description=Personal AI newsletter
#
#   [Service]
#   Type=oneshot
#   WorkingDirectory=/path/to/materiali-variante-1
#   ExecStart=/path/to/materiali-variante-1/.venv/bin/python newsletter.py
#
# File: /etc/systemd/system/newsletter.timer
#   [Unit]
#   Description=Run newsletter daily at 8:00
#
#   [Timer]
#   OnCalendar=*-*-* 08:00:00
#   Persistent=true
#
#   [Install]
#   WantedBy=timers.target
#
# Then: sudo systemctl enable --now newsletter.timer


# ====== NOTES ======
# - Check the venv Python path: which python (Mac/Linux) or where python (Windows)
# - The log in /tmp/newsletter.log lets you see what happened if the run fails
# - On Mac, give cron "Full Disk Access" in Privacy & Security
# - On Windows, if you want it to run even with PC suspended: check "Run whether user is logged on or not"
